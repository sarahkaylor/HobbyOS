#!/bin/bash
# ns.sh CMD... : run CMD (as the calling user) in a private PID + mount
# namespace.  The repo GUI harnesses run `pkill -9 -f qemu-system-*`; inside
# the namespace that can only reach this run's own QEMU.  With NS_TMP=<dir>,
# /tmp is a private bind of <dir> (harnesses hard-code /tmp paths).  Take the
# QEMU slot OUTSIDE: bash tools/cp/qemu-slot.sh ns.sh CMD...
U=$(id -u); G=$(id -g)
exec sudo -n env NS_TMP="$NS_TMP" NS_HOME="$HOME" NS_PATH="$PATH" NS_SLOT="${CP_QEMU_SLOT_HELD:-}" NS_MF="${MAKEFLAGS:-}" NS_TS="${TOOLS_SKIP:-}" unshare --kill-child --pid --fork --mount --propagation private --mount-proc bash -c '
  if [ -n "$NS_TMP" ]; then mkdir -p "$NS_TMP" && chown '$U:$G' "$NS_TMP" && mount --bind "$NS_TMP" /tmp || exit 99; fi
  exec setpriv --reuid='$U' --regid='$G' --init-groups -- env HOME="$NS_HOME" PATH="$NS_PATH" CP_QEMU_SLOT_HELD="$NS_SLOT" MAKEFLAGS="$NS_MF" TOOLS_SKIP="$NS_TS" "$@"' ns "$@"

#!/bin/bash
# C10-regress shim: clamp guest RAM to the strix slot cap (12 GiB), never let a
# GTK/SDL window reach the user's desktop session (-display gtk|sdl -> none,
# DISPLAY/WAYLAND_DISPLAY/XAUTHORITY unset), and log the exact command line.
# A GL display (egl-headless) is allowed ONLY as
# egl-headless,rendernode=/dev/dri/renderD129 and then runs with
# tools/cp/host-venus.sh's environment (its `env` output: venus-host-shim
# LD_PRELOAD + CP_SHIM_DENY of every other DRM node), integrator-approved.
unset DISPLAY WAYLAND_DISPLAY XAUTHORITY
LOG=${QSHIM_LOG:-/tmp/qshim.log}
REAL=${QSHIM_REAL:-/usr/bin/qemu-system-x86_64}
args=(); gl=0
while [ $# -gt 0 ]; do
  case "$1" in
    -m)
      v=$2; n=${v%%[MmGg]*}; u=${v#$n}
      case "$u" in G|g) mb=$((n*1024));; *) mb=$n;; esac
      [ "$mb" -gt 12288 ] && v=12288M
      args+=(-m "$v"); shift 2; continue ;;
    -display)
      v=$2
      case "$v" in
        gtk*|sdl*|cocoa*) v=none ;;
        egl-headless*)
          case ",$v," in
            *,rendernode=/dev/dri/renderD129,*) gl=1 ;;
            *) echo "[qshim] REFUSED: $v (GL only on rendernode=/dev/dri/renderD129)" >> "$LOG"; exit 3 ;;
          esac ;;
      esac
      args+=(-display "$v"); shift 2; continue ;;
  esac
  args+=("$1"); shift
done
HV=$HOME/cp/lanes/regress/tools/cp/host-venus.sh
if [ $gl = 1 ]; then
  export VENUS_BUILD=$HOME/cp/regress-work/venus RENDER=/dev/dri/renderD129
  bash "$HV" check >> "$LOG" 2>&1 || { echo "[qshim] REFUSED: host-venus.sh check failed" >> "$LOG"; exit 3; }
  mapfile -t E < <(bash "$HV" env)
  printf '[qshim] env %s\n' "${E[@]}" >> "$LOG"
  grep -q '^CP_SHIM_DENY=/dev/dri/' <(printf '%s\n' "${E[@]}") || { echo "[qshim] REFUSED: no CP_SHIM_DENY" >> "$LOG"; exit 3; }
  echo "[qshim] $(date +%T) GL qemu-system-x86_64 ${args[*]}" >> "$LOG"
  exec env -u DISPLAY -u WAYLAND_DISPLAY -u XAUTHORITY "${E[@]}" "$REAL" "${args[@]}"
fi
echo "[qshim] $(date +%T) qemu-system-x86_64 ${args[*]}" >> "$LOG"
exec "$REAL" "${args[@]}"

#!/bin/bash
# rwmon.sh <tree> <logfile> : while a GL RimWorld boot of <tree> runs, check
# every 5 s that its QEMU holds no DRM node except renderD129; on any other
# (renderD128, any card*) kill that QEMU at once (integrator condition).
T=$1; L=$2; seen=0
echo "rwmon start $(date +%T) tree=$T" >> "$L"
while :; do
  pid=""; for c in $(pgrep -f "qemu-system-x86_64.*egl-headless"); do [ "$(readlink /proc/$c/cwd)" = "$T" ] && pid=$c; done
  if [ -n "$pid" ]; then
    seen=1
    dri=$(ls -l /proc/$pid/fd 2>/dev/null | grep -o '/dev/dri/[a-zA-Z0-9]*' | sort | uniq -c | tr '\n' ' ')
    bad=$(ls -l /proc/$pid/fd 2>/dev/null | grep -o '/dev/dri/[a-zA-Z0-9]*' | grep -v '^/dev/dri/renderD129$' | head -1)
    echo "$(date +%T) pid=$pid dri: ${dri:-none}" >> "$L"
    envok=$(tr "\\0" "\\n" < /proc/$pid/environ 2>/dev/null | grep -c "^LD_PRELOAD=.*venus-host-shim.so$\\|^CP_SHIM_DENY=/dev/dri/")
    [ "$envok" = 2 ] || bad="no-shim-env($envok)"
    if [ -n "$bad" ]; then echo "$(date +%T) VIOLATION $bad -> kill -9 $pid" >> "$L"; kill -9 $pid; fi
  elif [ $seen = 1 ]; then echo "rwmon: qemu gone $(date +%T)" >> "$L"; exit 0; fi
  [ -f "$L.stop" ] && exit 0
  sleep 5
done

#!/usr/bin/env python3
"""ppmstats.py FILE.ppm... : per-frame stats for screendump evidence.
nonblack = fraction of pixels with any channel > 16; colours = distinct RGB
values over every 4th pixel; md5 = of the pixel payload (frames that differ
have different md5).  Prints one row per file and a summary."""
import hashlib, sys
def load(p):
    b = open(p, 'rb').read()
    # P6 header: magic, w, h, maxval separated by whitespace (no comments here)
    parts, i = [], 0
    while len(parts) < 4:
        while b[i:i+1].isspace(): i += 1
        j = i
        while not b[j:j+1].isspace(): j += 1
        parts.append(b[i:j]); i = j
    i += 1
    w, h = int(parts[1]), int(parts[2])
    return w, h, b[i:i + w*h*3]
rows = []
for p in sys.argv[1:]:
    w, h, px = load(p)
    n = w*h
    nb = sum(1 for k in range(0, len(px), 3) if px[k] > 16 or px[k+1] > 16 or px[k+2] > 16)
    cols = len({px[k:k+3] for k in range(0, len(px), 12)})
    md = hashlib.md5(px).hexdigest()[:12]
    rows.append((p, w, h, nb/n, cols, md))
    print(f"{p.split('/')[-1]:24s} {w}x{h} nonblack={nb/n:.3f} colours={cols} md5={md}")
print(f"frames={len(rows)} distinct={len({r[5] for r in rows})} "
      f"max_nonblack={max(r[3] for r in rows):.3f} max_colours={max(r[4] for r in rows)}")
