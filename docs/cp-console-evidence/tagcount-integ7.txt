# tagcount.py
#!/usr/bin/env python3
"""Per-tag line/byte counts of a HobbyOS serial log (tag = first [..] or word)."""
import re, sys, collections
for fn in sys.argv[1:]:
    data = open(fn, "rb").read().decode("utf-8", "replace")
    lines = data.split("\n")
    c = collections.Counter(); b = collections.Counter()
    for l in lines:
        s = l.lstrip().rstrip("\r")
        m = re.match(r"(\[[A-Za-z0-9_!\-]+\]|[A-Za-z]+)", s)
        k = m.group(1) if m else "(other)"
        c[k] += 1; b[k] += len(l) + 1
    tot = sum(b.values())
    print("==", fn, "lines", len(lines), "bytes", tot)
    for k, v in b.most_common(int(sys.argv[0] and 30)):
        print("%-24s %7d lines %9d bytes %5.1f%%" % (k, c[k], v, 100.0 * v / tot))
